CREATE OR REPLACE VIEW VW_MB_FRAUD_TRAIN AS
SELECT
    t.TRAN_ID,
    t.AC_ID,

    /* Amount Risk */
    (
        SELECT c.RISK_SCORE
        FROM MB_FRAUD_CRITERIA c
        WHERE c.CRITERIA_CODE = 'AMOUNT'
          AND t.TRAN_AMT_LC BETWEEN c.LOW_VALUE AND c.HIGH_VALUE
          AND c.ACTIVE_FLAG = 'Y'
    ) AS AMOUNT_RISK,

    /* Time Risk */
    (
        SELECT c.RISK_SCORE
        FROM MB_FRAUD_CRITERIA c
        WHERE c.CRITERIA_CODE = 'TIME'
          AND TO_NUMBER(TO_CHAR(t.CREATE_DATE,'HH24'))
              BETWEEN c.LOW_VALUE AND c.HIGH_VALUE
          AND c.ACTIVE_FLAG = 'Y'
    ) AS TIME_RISK,

    /* Velocity Risk */
    (
        SELECT c.RISK_SCORE
        FROM MB_FRAUD_CRITERIA c
        WHERE c.CRITERIA_CODE = 'VELOCITY'
          AND (
                SELECT COUNT(*)
                FROM EMOB.MB_TRANSACTION_DTL x
                WHERE x.AC_ID = t.AC_ID
                  AND TRUNC(x.CREATE_DATE) = TRUNC(t.CREATE_DATE)
              ) BETWEEN c.LOW_VALUE AND c.HIGH_VALUE
          AND c.ACTIVE_FLAG = 'Y'
    ) AS VELOCITY_RISK,

    /* Device Risk */
    CASE
        WHEN t.DEVICE_ID IS NOT NULL THEN 2
        ELSE 1
    END AS DEVICE_RISK,

    /* Debit / Credit Flag */
    CASE
        WHEN t.DRCR_CODE = 'D' THEN 1
        ELSE 0
    END AS DR_FLAG

FROM EMOB.MB_TRANSACTION_DTL t
WHERE t.TRAN_AMT_LC > 0
  AND t.CREATE_DATE >= (
        SELECT ADD_MONTHS(MAX(CREATE_DATE), -6)
        FROM EMOB.MB_TRANSACTION_DTL
      );



CREATE OR REPLACE PROCEDURE PRC_RETRAIN_FRAUD_MODEL AS
BEGIN
  BEGIN
    DBMS_DATA_MINING.DROP_MODEL('MB_TRANSACTION_FRAUD_MODEL');
  EXCEPTION WHEN OTHERS THEN NULL;
  END;

  DBMS_DATA_MINING.CREATE_MODEL(
    MODEL_NAME            => 'MB_TRANSACTION_FRAUD_MODEL',
    MINING_FUNCTION       => DBMS_DATA_MINING.CLASSIFICATION,
    DATA_TABLE_NAME       => 'VW_MB_FRAUD_TRAIN',
    CASE_ID_COLUMN_NAME   => 'TRAN_ID',
    TARGET_COLUMN_NAME    => NULL,
    SETTINGS_TABLE_NAME   => 'MB_SETTINGS'
  );
END;
